53void
54Kernel_Main(void)
55{

57 kernelAllocationMap.baseMem = &_brk;
58 kernelAllocationMap.brk = &_brk;
59 kernelAllocationMap.endMem = &_ramend;

61 Arch_InitIdleCpuMode();

63
64 if (LZ_CONFIG_MODULE_SERIAL_USED) {
65 Arch_InitSerial();

67
68 if (LZ_CONFIG_INSTRUMENT_CONTEXT_SWITCHES) {
69 Arch_InitInstrumentation();
70 }
71
72 /* Give hand to user */
73 main();
